# 无人机控制指令
## 一、UAVSetup 指令（设置类）

通过 `setup_pub` 发布，用于配置无人机状态。

|指令|功能|参数|说明|
|---|---|---|---|
|`ARM`|解锁电机|无|允许电机旋转，准备起飞|
|`DISARM`|上锁电机|无|禁止电机旋转|
|`SET_PX4_MODE`|设置PX4飞行模式|`px4_mode`|OFFBOARD / AUTO.LAND / POSCTL / AUTO.RTL|
|`REBOOT_PX4`|重启飞控|无|重新启动PX4固件|
|`SET_CONTROL_MODE`|设置上层控制模式|`control_mode`|CMD_CONTROL / RC_CONTROL / LAND_CONTROL|
|`EMERGENCY_KILL`|紧急停止|无|立即停止电机|

Python示例
```python
from sunray_msgs.msg import UAVSetup

# 创建指令
setup = UAVSetup()
setup.header.stamp = rospy.Time.now()

# 1. 解锁
setup.cmd = UAVSetup.ARM
pub.publish(setup)

# 2. 上锁
setup.cmd = UAVSetup.DISARM
pub.publish(setup)

# 3. 设置PX4模式为OFFBOARD
setup.cmd = UAVSetup.SET_PX4_MODE
setup.px4_mode = "OFFBOARD"
pub.publish(setup)

# 4. 设置控制模式为CMD_CONTROL
setup.cmd = UAVSetup.SET_CONTROL_MODE
setup.control_mode = "CMD_CONTROL"
pub.publish(setup)

# 5. 重启飞控
setup.cmd = UAVSetup.REBOOT_PX4
pub.publish(setup)

# 6. 紧急停止
setup.cmd = UAVSetup.EMERGENCY_KILL
pub.publish(setup)
```

## 二、UAVControlCMD 指令（控制类）

通过 `cmd_pub` 发布，用于控制无人机飞行。

### 1. 位置控制指令

|指令|功能|坐标系|参数|
|---|---|---|---|
|`XyzPos`|位置控制|惯性系|`desired_pos[0/1/2]`|
|`XyzPosYaw`|位置+偏航|惯性系|`desired_pos[0/1/2]` + `desired_yaw`|
|`XyzPosYawBody`|位置+偏航|机体坐标系|`desired_pos[0/1/2]`(前/右/上) + `desired_yaw`|
|`GlobalPos`|全局位置|WGS84|`latitude` + `longitude` + `altitude` + `desired_yaw`|

### 2. 飞行状态指令

|指令|功能|
|---|---|
|`Takeoff`|起飞|
|`Land`|降落|
|`Hover`|悬停|
|`Return`|返航|
|`Point`|路径规划目标点|

Python演示
```python
from sunray_msgs.msg import UAVControlCMD
import math

# 创建指令
cmd = UAVControlCMD()
cmd.header.stamp = rospy.Time.now()

# 1. 惯性系位置控制（保持当前偏航）
cmd.cmd = UAVControlCMD.XyzPos
cmd.desired_pos[0] = 5.0   # X坐标（米）
cmd.desired_pos[1] = 3.0   # Y坐标（米）
cmd.desired_pos[2] = 2.0   # Z坐标（米）

# 2. 惯性系位置+偏航控制
cmd.cmd = UAVControlCMD.XyzPosYaw
cmd.desired_pos[0] = 5.0
cmd.desired_pos[1] = 3.0
cmd.desired_pos[2] = 2.0
cmd.desired_yaw = 45.0 * math.pi / 180.0  # 偏航角（弧度）

# 3. 机体坐标系位置控制
cmd.cmd = UAVControlCMD.XyzPosYawBody
cmd.desired_pos[0] = 5.0   # 前向位移（米）
cmd.desired_pos[1] = 2.0   # 右向位移（米）
cmd.desired_pos[2] = 1.0   # 向上位移（米）
cmd.desired_yaw = 0.0

# 4. 全局位置控制
cmd.cmd = UAVControlCMD.GlobalPos
cmd.latitude = 31.2345      # 纬度（度）
cmd.longitude = 121.4567    # 经度（度）
cmd.altitude = 50.0         # 海拔高度（米）
cmd.desired_yaw = 0.0

# 5. 起飞
cmd.cmd = UAVControlCMD.Takeoff

# 6. 降落
cmd.cmd = UAVControlCMD.Land

# 7. 悬停
cmd.cmd = UAVControlCMD.Hover

# 8. 返航
cmd.cmd = UAVControlCMD.Return

# 9. 路径规划目标点
cmd.cmd = UAVControlCMD.Point
cmd.desired_pos[0] = 10.0
cmd.desired_pos[1] = 10.0
cmd.desired_pos[2] = 5.0
cmd.desired_yaw = 0.0

# 发布
pub.publish(cmd)
```

## 三、UAVState 指令（状态订阅）

通过订阅 `uav_state` 话题获取无人机状态。

|字段|类型|说明|
|---|---|---|
|`connected`|bool|是否已连接|
|`armed`|bool|是否已解锁|
|`control_mode`|string|控制模式（CMD_CONTROL / RC_CONTROL 等）|
|`px4_mode`|string|PX4模式（OFFBOARD / POSCTL 等）|
|`landed_state`|int|着陆状态（0=未知，1=着陆，2=空中）|
|`local_position`|vector|本地坐标位置|
|`local_velocity`|vector|本地速度|
|`attitude`|vector|姿态角（滚转、俯仰、偏航）|

Python演示
```python
from sunray_msgs.msg import UAVState

# 状态回调函数
def state_callback(msg):
    print(f"连接状态: {msg.connected}")
    print(f"解锁状态: {msg.armed}")
    print(f"控制模式: {msg.control_mode}")
    print(f"PX4模式: {msg.px4_mode}")
    print(f"着陆状态: {msg.landed_state}")  # 0=未知,1=着陆,2=空中
    print(f"当前位置: x={msg.local_position[0]}, y={msg.local_position[1]}, z={msg.local_position[2]}")
    print(f"当前速度: vx={msg.local_velocity[0]}, vy={msg.local_velocity[1]}, vz={msg.local_velocity[2]}")
    print(f"姿态角: roll={msg.attitude[0]}, pitch={msg.attitude[1]}, yaw={msg.attitude[2]}")

# 订阅状态话题
rospy.Subscriber('/uav1/sunray/uav_state', UAVState, state_callback)
```

## 四、完整使用示例
```python
import rospy
import math
from sunray_msgs.msg import UAVSetup, UAVControlCMD, UAVState

class DroneController:
    def __init__(self, uav_id=1):
        rospy.init_node('drone_controller', anonymous=True)
        self.uav_name = f"/uav{uav_id}"
        
        # 创建发布器
        self.setup_pub = rospy.Publisher(f"{self.uav_name}/sunray/setup", UAVSetup, queue_size=1)
        self.cmd_pub = rospy.Publisher(f"{self.uav_name}/sunray/uav_control_cmd", UAVControlCMD, queue_size=1)
        
        # 订阅状态
        rospy.Subscriber(f"{self.uav_name}/sunray/uav_state", UAVState, self.state_callback)
        
        self.armed = False
        self.control_mode = ""
        self.px4_mode = ""
        self.landed_state = 0
        
        rospy.sleep(1)
        rospy.loginfo(f"初始化完成 - {self.uav_name}")
    
    def state_callback(self, msg):
        self.armed = msg.armed
        self.control_mode = msg.control_mode
        self.px4_mode = msg.px4_mode
        self.landed_state = msg.landed_state
    
    def arm(self):
        """解锁"""
        setup = UAVSetup()
        setup.header.stamp = rospy.Time.now()
        setup.cmd = UAVSetup.ARM
        self.setup_pub.publish(setup)
        rospy.loginfo("解锁指令已发送")
    
    def disarm(self):
        """上锁"""
        setup = UAVSetup()
        setup.header.stamp = rospy.Time.now()
        setup.cmd = UAVSetup.DISARM
        self.setup_pub.publish(setup)
        rospy.loginfo("上锁指令已发送")
    
    def set_offboard_mode(self):
        """切换到OFFBOARD模式"""
        setup = UAVSetup()
        setup.header.stamp = rospy.Time.now()
        setup.cmd = UAVSetup.SET_PX4_MODE
        setup.px4_mode = "OFFBOARD"
        self.setup_pub.publish(setup)
        rospy.loginfo("OFFBOARD模式切换指令已发送")
    
    def set_cmd_control(self):
        """切换到指令控制模式"""
        setup = UAVSetup()
        setup.header.stamp = rospy.Time.now()
        setup.cmd = UAVSetup.SET_CONTROL_MODE
        setup.control_mode = "CMD_CONTROL"
        self.setup_pub.publish(setup)
        rospy.loginfo("CMD_CONTROL模式切换指令已发送")
    
    def takeoff(self):
        """起飞"""
        cmd = UAVControlCMD()
        cmd.header.stamp = rospy.Time.now()
        cmd.cmd = UAVControlCMD.Takeoff
        self.cmd_pub.publish(cmd)
        rospy.loginfo("起飞指令已发送")
    
    def land(self):
        """降落"""
        cmd = UAVControlCMD()
        cmd.header.stamp = rospy.Time.now()
        cmd.cmd = UAVControlCMD.Land
        self.cmd_pub.publish(cmd)
        rospy.loginfo("降落指令已发送")
    
    def goto(self, x, y, z, yaw_deg=0):
        """飞到指定位置"""
        cmd = UAVControlCMD()
        cmd.header.stamp = rospy.Time.now()
        cmd.cmd = UAVControlCMD.XyzPosYaw
        cmd.desired_pos[0] = x
        cmd.desired_pos[1] = y
        cmd.desired_pos[2] = z
        cmd.desired_yaw = yaw_deg * math.pi / 180.0
        self.cmd_pub.publish(cmd)
        rospy.loginfo(f"位置指令已发送: ({x}, {y}, {z})m, 偏航{yaw_deg}°")

# 主程序
if __name__ == "__main__":
    drone = DroneController(uav_id=1)
    rospy.sleep(2)
    
    # 完整起飞流程
    drone.set_cmd_control()   # 1. 切换控制模式
    rospy.sleep(1)
    drone.set_offboard_mode() # 2. 切换PX4模式
    rospy.sleep(1)
    drone.arm()               # 3. 解锁
    rospy.sleep(2)
    drone.takeoff()           # 4. 起飞
    rospy.sleep(5)
    drone.goto(5, 0, 10, 45)  # 5. 飞到指定位置
    
    rospy.spin()
```

## 五、关键要点

|项目|说明|
|---|---|
|**时间戳**|所有指令必须设置 `header.stamp = rospy.Time.now()`|
|**偏航角单位**|弧度（使用 `math.pi / 180.0` 转换）|
|**发布前提**|控制指令需要已解锁 + OFFBOARD 模式|
|**持续发送**|OFFBOARD 模式下需持续发送控制指令（建议 10Hz）|
|**坐标系**|NED：X=北，Y=东，Z=天|
